// ==================== TMDB CONFIG ==================== const API_KEY = '7a98db423d6e3a5ee922a3e51a09d135'; const BASE_URL = 'https://api.themoviedb.org/3'; const IMG_URL = 'https://image.tmdb.org/t/p/w500'; const BACKDROP_URL = 'https://image.tmdb.org/t/p/original'; // ==================== LISTA E FILMAVE DEMO ==================== // Ti do t'i zëvendësosh këto ID me ato të filmave të tu const MY_MOVIES = [27205, 155, 49026, 238, 680]; // Inception, Dark Knight, Interstellar, Godfather, Pulp Fiction const MY_TV_SHOWS = [1396, 1399, 106459]; // Breaking Bad, Stranger Things, Narcos // ==================== FUNKSIONET ==================== async function fetchMovieById(id) { const res = await fetch(`${BASE_URL}/movie/${id}?api_key=${API_KEY}&language=en-US`); const data = await res.json(); return { id: data.id, title: data.title, year: data.release_date ? data.release_date.split('-')[0] : 'N/A', rating: data.vote_average ? data.vote_average.toFixed(1) : 'N/A', poster: data.poster_path ? IMG_URL + data.poster_path : 'https://placehold.co/500x750/1a1a1a/e50914?text=No+Poster', overview: data.overview || 'No description available.', genres: data.genres ? data.genres.map(g => g.name).join(', ') : 'N/A' }; } async function fetchTvById(id) { const res = await fetch(`${BASE_URL}/tv/${id}?api_key=${API_KEY}&language=en-US`); const data = await res.json(); return { id: data.id, title: data.name, year: data.first_air_date ? data.first_air_date.split('-')[0] : 'N/A', rating: data.vote_average ? data.vote_average.toFixed(1) : 'N/A', poster: data.poster_path ? IMG_URL + data.poster_path : 'https://placehold.co/500x750/1a1a1a/e50914?text=No+Poster', overview: data.overview || 'No description available.', genres: data.genres ? data.genres.map(g => g.name).join(', ') : 'N/A' }; } function renderRow(containerId, items) { const container = document.getElementById(containerId); if (!container) return; container.innerHTML = items.map(item => `

${item.title} (${item.year})

⭐ ${item.rating}/10

`).join(''); } async function loadMovies() { // Trending (të parët nga MY_MOVIES) const trendingMovies = await Promise.all(MY_MOVIES.slice(0, 6).map(id => fetchMovieById(id))); trendingMovies.forEach(m => m.type = 'movie'); renderRow('trendingRow', trendingMovies); // Movies const movies = await Promise.all(MY_MOVIES.map(id => fetchMovieById(id))); movies.forEach(m => m.type = 'movie'); renderRow('moviesRow', movies); // TV Shows const tvShows = await Promise.all(MY_TV_SHOWS.map(id => fetchTvById(id))); tvShows.forEach(t => t.type = 'tv'); renderRow('tvRow', tvShows); } // MODAL FUNCTIONS function openMovieModal(id, type) { const modal = document.getElementById('movieModal'); document.getElementById('modalTitle').innerText = 'Loading...'; document.getElementById('modalOverview').innerText = ''; document.getElementById('modalImage').src = 'https://placehold.co/500x750/1a1a1a/e50914?text=Loading'; modal.style.display = 'flex'; const fetchFunc = type === 'tv' ? fetchTvById : fetchMovieById; fetchFunc(id).then(item => { document.getElementById('modalTitle').innerText = item.title; document.getElementById('modalYear').innerHTML = ` ${item.year}`; document.getElementById('modalRating').innerHTML = ` ${item.rating}/10`; document.getElementById('modalOverview').innerText = item.overview; document.getElementById('modalGenres').innerHTML = ` ${item.genres}`; document.getElementById('modalImage').src = item.poster; }).catch(err => { document.getElementById('modalTitle').innerText = 'Error loading movie'; console.error(err); }); } function closeModal() { document.getElementById('movieModal').style.display = 'none'; } // KËRKIMI async function searchTMDB(query) { if (!query || query.length < 2) { loadMovies(); return; } const res = await fetch(`${BASE_URL}/search/multi?api_key=${API_KEY}&query=${encodeURIComponent(query)}`); const data = await res.json(); const results = data.results.slice(0, 10).map(item => ({ id: item.id, title: item.title || item.name, year: (item.release_date || item.first_air_date || '').split('-')[0] || 'N/A', rating: item.vote_average ? item.vote_average.toFixed(1) : 'N/A', poster: item.poster_path ? IMG_URL + item.poster_path : 'https://placehold.co/500x750/1a1a1a/e50914?text=No+Poster', type: item.media_type === 'tv' ? 'tv' : 'movie' })); renderRow('trendingRow', results); } // EVENTET document.addEventListener('DOMContentLoaded', () => { loadMovies(); const searchInput = document.getElementById('searchInput'); if (searchInput) { let timeout; searchInput.addEventListener('input', (e) => { clearTimeout(timeout); timeout = setTimeout(() => searchTMDB(e.target.value.trim()), 500); }); } }); // Ekspozo funksionet globalisht window.openMovieModal = openMovieModal; window.closeModal = closeModal;
